# P11 - CHANGED CODE
# Prerequisite: VARIABLES; work exists on your feature branch and relevant checks have passed.
# This flow does not switch branches. Check BR_NAME against the actual branch first.
# Run after each meaningful change; choose a fresh COMMIT_MSG. Then use PUBLISH.
# ==============================================================================

# P11 - Step 01: Enter checkout.
# Bash counterpart: B11, same purpose; cmdlet help: Get-Help <name> -Examples.
Set-Location -LiteralPath $env:PROJECT_ROOT

# P11 - Step 02: Verify branch matches BR_NAME and inspect pending files.
# Bash counterpart: B11, same purpose; cmdlet help: Get-Help <name> -Examples.
git.exe status --short --branch

# P11 - Step 03: Review tracked changes. New files are reviewed after staging as well.
# Bash counterpart: B11, same purpose; cmdlet help: Get-Help <name> -Examples.
git.exe diff

# P11 - Step 04: Stage all reviewed tracked changes from repository root.
# Use only when all tracked modifications belong to this commit. For partial scope, use git add --interactive.
# New untracked files require explicit git add -- path/to/file separately.
git.exe add -u

# P11 - Step 05: Review the complete staged patch; correct unwanted staging before committing.
# Bash counterpart: B11, same purpose; cmdlet help: Get-Help <name> -Examples.
git.exe diff --cached

# P11 - Step 06: Check staged whitespace. No output/exit 0 means pass.
# Bash counterpart: B11, same purpose; cmdlet help: Get-Help <name> -Examples.
git.exe diff --cached --check

# P11 - Step 07: Commit reviewed scope using the chosen message.
# Bash counterpart: B11, same purpose; cmdlet help: Get-Help <name> -Examples.
git.exe commit -m $env:COMMIT_MSG

# P11 - Step 08: Confirm remaining work; intentionally unstaged changes may remain.
# Bash counterpart: B11, same purpose; cmdlet help: Get-Help <name> -Examples.
git.exe status --short --branch

# P11 - Step 09: Capture commit evidence.
# Bash counterpart: B11, same purpose; cmdlet help: Get-Help <name> -Examples.
git.exe log -1 --oneline

# P08 - Step 08: Return to integration branch.
# Bash counterpart: B08, same purpose; cmdlet help: Get-Help <name> -Examples.

# P08 - Step 09: Integrate remote merge.

# P13 - AZ ACCOUNT
# Optional account-context flow; no resource provisioning. Requires an installed, authenticated Azure CLI.
# If not authenticated, stop for the Azure setup lesson. No automatic login or resource creation.
# Bash: if command resolves under /mnt/c, it is Windows CLI; stop rather than mixing credentials/runtimes.
# Select subscription deliberately; account set changes CLI default context. Never paste tokens.
# ==============================================================================

# P13 - Step 01: Locate CLI before use. PowerShell help: Get-Help Get-Command -Examples.
# Bash counterpart: B13, same purpose; cmdlet help: Get-Help <name> -Examples.
Get-Command az -All

# P13 - Step 02: Inspect installed version.
# Bash counterpart: B13, same purpose; cmdlet help: Get-Help <name> -Examples.
az version

# P13 - Step 03: List subscriptions available to current login.
# Bash counterpart: B13, same purpose; cmdlet help: Get-Help <name> -Examples.
az account list --output table

# P13 - Step 04: Set chosen subscription identifier once; replace placeholder.
# Bash counterpart: B13, same purpose; cmdlet help: Get-Help <name> -Examples.
$env:AZ_SUBSCRIPTION = 'replace_with_subscription_id'

# P13 - Step 05: Select the intended subscription.
# Bash counterpart: B13, same purpose; cmdlet help: Get-Help <name> -Examples.
az account set --subscription $env:AZ_SUBSCRIPTION

# P13 - Step 06: Verify selected account context.
# Bash counterpart: B13, same purpose; cmdlet help: Get-Help <name> -Examples.
az account show --output table
