# B00 - VARIABLES
# Run once per terminal; edit values for the task. Each shell has its own environment.

# B00 - Step 01: Set canonical checkout. PowerShell alternative: an already-established native Windows checkout.
export PROJECT_ROOT='/home/amola/projects/staff_de_training'

# B00 - Step 02: Choose feature branch; replace this example once.

# B00 - Step 04: Choose commit message; update for each logical change.

# B12 - NEW CODE

# B12 - Step 01: Enter checkout.

# B11 - CHANGED CODE
# Prerequisite: VARIABLES; work exists on your feature branch and relevant checks have passed.
# This flow does not switch branches. Check BR_NAME against the actual branch first.
# Run after each meaningful change; choose a fresh COMMIT_MSG. Then use PUBLISH.
# ==============================================================================

# B11 - Step 01: Enter checkout.
cd "$PROJECT_ROOT"

# B11 - Step 02: Verify branch matches BR_NAME and inspect pending files.
git status --short --branch

# B11 - Step 03: Review tracked changes. New files are reviewed after staging as well.
git diff

# B11 - Step 04: Stage all reviewed tracked changes from repository root.
# Use only when all tracked modifications belong to this commit. For partial scope, use git add --interactive.
# New untracked files require explicit git add -- path/to/file separately.
git add -u

# B11 - Step 05: Review the complete staged patch; correct unwanted staging before committing.
git diff --cached

# B11 - Step 06: Check staged whitespace. No output/exit 0 means pass.
git diff --cached --check

# B11 - Step 07: Commit reviewed scope using the chosen message.
git commit -m "$COMMIT_MSG"

# B11 - Step 08: Confirm remaining work; intentionally unstaged changes may remain.
git status --short --branch

# B11 - Step 09: Capture commit evidence.
git log -1 --oneline

# B08 - PUBLISH
# Prerequisite: reviewed feature commit; BR_NAME names the current branch. Owner-run remote actions.
# Missing checks is not CI acceptance. Do not merge while required checks/review are unresolved.
# ==============================================================================

# B08 - Step 01: Enter checkout.
cd "$PROJECT_ROOT"

# B08 - Step 02: Verify branch and publication scope.
git status --short --branch

# B08 - Step 03: Push selected feature branch and set upstream.
git push -u origin "$BR_NAME"

# B08 - Step 04: Open PR interactively; describe problem, behavior, checks and limitations. Skip if PR already exists.
gh pr create --base "$BASE_BRANCH" --head "$BR_NAME"

# B08 - Step 05: Inspect PR.
gh pr view "$BR_NAME"

# B08 - Step 06: Inspect hosted checks; address failures. Merge in GitHub after review, then continue.
gh pr checks "$BR_NAME"

# B08 - Step 07: After hosted merge, verify worktree clean before switching.
git status --short --branch

# B08 - Step 08: Return to integration branch.
git switch "$BASE_BRANCH"

# B08 - Step 09: Integrate remote merge.
git pull --ff-only

# B08 - Step 10: Verify integrated change in history.
git log -5 --oneline

# B08 - Step 11: Delete integrated local branch. If squash history makes -d refuse, inspect rather than force.
git branch -d "$BR_NAME"

# B03 - SNAPSHOT
# Prerequisite: VARIABLES and a reviewed, working environment. Intentionally replaces SNAPSHOT_FILE.
# requirements.txt is handwritten direct intent; snapshot records installed packages, not a solver lock.
# Use separate Windows/Linux snapshots. Review new output before accepting it.

# B05 - VALIDATE
# Prerequisite: VARIABLES, REPRODUCE and PROFILE completed for this platform.
# Step 01 is connection/configuration only. Step 02 WRITES selected model relations.
# Build requires existing raw.orders in the selected database; a new empty database is not ready.
# Windows full build remains gated on native loader/path acceptance; see PIPELINE.
# ==============================================================================

# B05 - Step 01: Validate connection with recreated runtime and template; expect All checks passed.
"$REPRO_DIR/bin/dbt" debug --project-dir "$DBT_PROJECT_DIR" --profiles-dir "$PROFILE_DIR" --target-path "$DBT_TARGET_PATH" --log-path "$DBT_LOG_PATH"

# B05 - Step 02: AFTER raw source readiness: build descendants and tests. Freshness is a separate command.
"$REPRO_DIR/bin/dbt" build --project-dir "$DBT_PROJECT_DIR" --profiles-dir "$PROFILE_DIR" --select "source:commerce.orders+" --target-path "$DBT_TARGET_PATH" --log-path "$DBT_LOG_PATH"

# B07 - PIPELINE
# Prerequisite: VARIABLES and PROFILE. Deliberate writes, not a startup block.
# Current loader replaces raw.orders and stamps loaded_at. It ignores DBT_DUCKDB_PATH.
# Bash: DBT_DUCKDB_PATH MUST equal DBT_PROJECT_DIR/dev.duckdb before loading.
# Native Windows: STOP at inspection; loader currently targets the shared script-adjacent database.
# Freshness is recency, not completeness. Stop on failed commands; no automatic gating is implied.
# ==============================================================================

# B07 - Step 01: Inspect loader path contract. PowerShell help: Get-Help Get-Content -Examples.
cat "$DBT_PROJECT_DIR/load_orders.py"

# B07 - Step 02: Verify loader/dbt target agreement; Bash exit 1 means STOP.
test "$DBT_DUCKDB_PATH" = "$DBT_PROJECT_DIR/dev.duckdb"

# B07 - Step 03: Replace raw data with fictional fixture.
"$PYTHON_EXE" "$DBT_PROJECT_DIR/load_orders.py"

# B07 - Step 04: Check freshness; inspect warnings and stop on error.
"$DBT_EXE" source freshness --project-dir "$DBT_PROJECT_DIR" --profiles-dir "$PROFILE_DIR" --target-path "$DBT_TARGET_PATH" --log-path "$DBT_LOG_PATH"

# B07 - Step 05: Build selected models and tests.
"$DBT_EXE" build --project-dir "$DBT_PROJECT_DIR" --profiles-dir "$PROFILE_DIR" --select "source:commerce.orders+" --target-path "$DBT_TARGET_PATH" --log-path "$DBT_LOG_PATH"
